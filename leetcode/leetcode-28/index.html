class Solution:
    def strStr(self, haystack: str, needle: str) -> int:
        first_need = needle[0]
        num = 0

        for i in range(len(haystack) - len(needle) + 1):
            if haystack[i] == first_need:
                for j in range(len(needle)):
                    if haystack[i + j] != needle[j]:
                        break
                    else:
                        num += 1

                if num == len(needle):
                    return i

                num = 0

        return -1

class Solution:
    def strStr(self, haystack: str, needle: str) -> int:
        n = len(haystack)
        m = len(needle)

        for i in range(n - m + 1):
            matched = True

            for j in range(m):
                if haystack[i + j] != needle[j]:
                    matched = False
                    break

            if matched:
                return i

        return -1

class Solution:
    def strStr(self, haystack: str, needle: str) -> int:
        n = len(haystack)
        m = len(needle)

        lps = [0] * m

        j = 0
        for i in range(1, m):
            while j > 0 and needle[i] != needle[j]:
                j = lps[j - 1]

            if needle[i] == needle[j]:
                j += 1

            lps[i] = j

        j = 0

        for i in range(n):
            while j > 0 and haystack[i] != needle[j]:
                j = lps[j - 1]

            if haystack[i] == needle[j]:
                j += 1

            if j == m:
                return i - m + 1

        return -1
