class Solution:
    def addBinary(self, a: str, b: str) -> str:
        # 补齐到相同长度（左边补0），这样下标就能对齐
        n = max(len(a), len(b))
        a = a.zfill(n)
        b = b.zfill(n)

        # 结果可能多一位进位，所以长度是 n+1
        s = ['0'] * (n + 1)
        count = 0  # 进位

        # 从右往左遍历
        for i in range(n - 1, -1, -1):
            # 注意：a[i] + b[i] 是字符串，要先转 int
            total = int(a[i]) + int(b[i])

            if count == 0:
                if total == 0:
                    s[i + 1] = '0'
                elif total == 1:
                    s[i + 1] = '1'
                elif total == 2:
                    s[i + 1] = '0'
                    count = 1
            else:
                if total == 0:
                    s[i + 1] = '1'
                    count = 0
                elif total == 1:
                    s[i + 1] = '0'
                    count = 1
                elif total == 2:
                    s[i + 1] = '1'
                    count = 1

        # 处理最高位的进位
        if count == 1:
            s[0] = '1'
            return ''.join(s)
        else:
            return ''.join(s[1:])

class Solution:
    def addBinary(self, a: str, b: str) -> str:
        i = len(a) - 1
        j = len(b) - 1
        carry = 0
        ans = []

        while i >= 0 or j >= 0 or carry:
            total = carry

            if i >= 0:
                total += int(a[i])
                i -= 1

            if j >= 0:
                total += int(b[j])
                j -= 1

            ans.append(str(total % 2))
            carry = total // 2

        return ''.join(reversed(ans))

然后：

1

1

执行 L 次。 每一次里面只有：

1

4

加法
比较

这些操作都可以看成 O(1)。 所以循环是：

1
2

= O(L)

最后的：

1

需要连接大约 L 个字符，也是：

1

O(L)

总时间：

1
2

O(L) + O(L) + O(L) + O(L)

1

O(L)

所以我的时间复杂度是：

空间方面，zfill() 产生新字符串，s 也会随着输入长度增长，所以：
